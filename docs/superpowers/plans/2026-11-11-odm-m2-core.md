- [ ] **Step 3: Implement the limiter**

Create `crates/odm-core/src/limiter.rs`:

```rust
pub struct TokenBucket { /* rate: u64, state: Mutex<BucketState> */ }

impl TokenBucket {
    pub fn new(bytes_per_second: u64) -> Self;
    pub fn unlimited() -> Self; // rate 0
    pub async fn acquire(&self, bytes: u64);
}

#[derive(Clone)]
pub struct Limiter { /* global: Arc<TokenBucket>, job: Option<Arc<TokenBucket>> */ }

impl Limiter {
    pub fn new(global: Arc<TokenBucket>, job_limit: Option<u64>) -> Self;
    pub async fn acquire(&self, bytes: u64);
}
```

Behavior rules:

- A rate of `0` means unlimited; `acquire` returns immediately.
- The bucket starts empty. Each `acquire(n)` takes `n` tokens; if fewer are available it sleeps for `(n - tokens) / rate` seconds and retries. Tokens refill at `rate` per second, capped at `max(rate, n)` for that call so a chunk larger than one second of budget cannot deadlock.
- Use `tokio::time::{Instant, sleep}` so paused-time tests work.
- Use `std::sync::Mutex` and recover a poisoned lock with `unwrap_or_else(|poisoned| poisoned.into_inner())`; no `unwrap()`.
- `Limiter::new` creates a job bucket only for `Some(rate)` with `rate > 0`; `None` and `Some(0)` mean no per-job limit.
- `Limiter::acquire` awaits the global bucket first, then the job bucket.

In `job.rs`, add to `JobOptions` and put `#[serde(default)]` on the struct:

```rust
pub segments: Option<u32>,
pub speed_limit: Option<u64>,
```

`JobOptions::default()` leaves both `None`. Older persisted JSON must still deserialize.

Update `lib.rs` to declare `pub mod limiter;` and re-export `Limiter` and `TokenBucket`.

//! Segmented transfer, resume, retry and split tests against a local server.
#![allow(clippy::unwrap_used, clippy::expect_used)]

use std::net::SocketAddr;
use std::path::Path;
use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::{Arc, Mutex};
use std::time::Duration;

use axum::Router;
use axum::body::Body;
use axum::extract::State;
use axum::http::{HeaderMap, HeaderValue, Method, StatusCode, header};
use axum::response::{IntoResponse, Response};
use axum::routing::any;
use futures_util::stream;
use odm_core::{
    Backend, BackendKind, ControlToken, HttpBackend, Job, JobContext, JobOptions, Limiter,
    ProbeResult, ProgressSink, RunOutcome, Segment, Source, Store, TokenBucket,
};
use tempfile::TempDir;
use tokio::net::TcpListener;

const BLOCK: usize = 64 * 1024;

fn payload(size: usize) -> Arc<Vec<u8>> {
    Arc::new((0..size).map(|index| (index % 251) as u8).collect())
}

struct TestServer {
    addr: SocketAddr,
    state: Arc<ServerState>,
    handle: tokio::task::JoinHandle<()>,
}

struct ServerState {
    payload: Vec<u8>,
    etag: Mutex<String>,
    supports_ranges: bool,
    slow: bool,
    fail_next: AtomicUsize,
    ranges: Mutex<Vec<String>>,
    if_ranges: Mutex<Vec<String>>,
}

impl TestServer {
    async fn start(payload: Vec<u8>, supports_ranges: bool, slow: bool) -> Self {
        let state = Arc::new(ServerState {
            payload,
            etag: Mutex::new("\"v1\"".to_string()),
            supports_ranges,
            slow,
            fail_next: AtomicUsize::new(0),
            ranges: Mutex::new(Vec::new()),
            if_ranges: Mutex::new(Vec::new()),
        });
        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
        let addr = listener.local_addr().unwrap();
        let app = Router::new()
            .route("/file.bin", any(serve))
            .with_state(state.clone());
        let handle = tokio::spawn(async move {
            axum::serve(listener, app).await.unwrap();
        });
        Self {
            addr,
            state,
            handle,
        }
    }

    fn url(&self) -> String {
        format!("http://{}/file.bin", self.addr)
    }

    fn set_etag(&self, etag: &str) {
        *self.state.etag.lock().unwrap() = etag.to_string();
    }
}

impl Drop for TestServer {
    fn drop(&mut self) {
        self.handle.abort();
    }
}

async fn serve(
    method: Method,
    headers: HeaderMap,
    State(state): State<Arc<ServerState>>,
) -> Response {
    if method == Method::HEAD {
        let mut response = HeaderMap::new();
        response.insert(
            header::CONTENT_LENGTH,
            HeaderValue::from_str(&state.payload.len().to_string()).unwrap(),
        );
        response.insert(
            header::ETAG,
            HeaderValue::from_str(&state.etag.lock().unwrap()).unwrap(),
        );
        response.insert(
            header::CONTENT_TYPE,
            HeaderValue::from_static("application/octet-stream"),
        );
        if state.supports_ranges {
            response.insert(header::ACCEPT_RANGES, HeaderValue::from_static("bytes"));
        }
        return (StatusCode::OK, response).into_response();
    }

    if let Some(range) = headers
        .get(header::RANGE)
        .and_then(|value| value.to_str().ok())
    {
        state.ranges.lock().unwrap().push(range.to_string());
    }
    if let Some(if_range) = headers
        .get(header::IF_RANGE)
        .and_then(|value| value.to_str().ok())
    {
        state.if_ranges.lock().unwrap().push(if_range.to_string());
    }
    if state
        .fail_next
        .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |value| {
            value.checked_sub(1)
        })
        .is_ok()
    {
        return StatusCode::INTERNAL_SERVER_ERROR.into_response();
    }

    let range = headers
        .get(header::RANGE)
        .and_then(|value| value.to_str().ok())
        .and_then(|value| value.strip_prefix("bytes="))
        .and_then(|value| value.split_once('-'))
        .and_then(|(start, end)| Some((start.parse::<usize>().ok()?, end.parse::<usize>().ok()?)));

    let etag = state.etag.lock().unwrap().clone();
    if let Some(if_range) = headers
        .get(header::IF_RANGE)
        .and_then(|value| value.to_str().ok())
    {
        if if_range != etag {
            return full(&state, StatusCode::OK);
        }
    }

    match range {
        Some((start, end)) if state.supports_ranges => {
            let end = end.min(state.payload.len() - 1);
            let slice = state.payload[start..=end].to_vec();
            let mut response = HeaderMap::new();
            response.insert(
                header::CONTENT_RANGE,
                HeaderValue::from_str(&format!(
                    "bytes {start}-{end}/{}",
                    state.payload.len()
                ))
                .unwrap(),
            );
            response.insert(
                header::CONTENT_LENGTH,
                HeaderValue::from_str(&slice.len().to_string()).unwrap(),
            );
            response.insert(header::ETAG, HeaderValue::from_str(&etag).unwrap());
            let delay = if state.slow && start == 0 { 5 } else { 1 };
            (
                StatusCode::PARTIAL_CONTENT,
                response,
                body(slice, delay),
            )
                .into_response()
        }
        _ => full(&state, StatusCode::OK),
    }
}

fn full(state: &ServerState, status: StatusCode) -> Response {
    let mut response = HeaderMap::new();
    response.insert(
        header::CONTENT_LENGTH,
        HeaderValue::from_str(&state.payload.len().to_string()).unwrap(),
    );
    response.insert(
        header::ETAG,
        HeaderValue::from_str(&state.etag.lock().unwrap()).unwrap(),
    );
    (
        status,
        response,
        body(state.payload.clone(), if state.slow { 5 } else { 0 }),
    )
        .into_response()
}

fn body(bytes: Vec<u8>, delay_ms: u64) -> Body {
    if delay_ms == 0 {
        return Body::from(bytes);
    }

    let chunks = stream::unfold(bytes, move |mut bytes| async move {
        if bytes.is_empty() {
            return None;
        }
        let rest = bytes.split_off(bytes.len().min(BLOCK));
        tokio::time::sleep(Duration::from_millis(delay_ms)).await;
        Some((Ok::<Vec<u8>, std::io::Error>(bytes), rest))
    });
    Body::from_stream(chunks)
}

In `http.rs`, keep the M1 single-stream path for sources without ranges or size. Add:

```rust
const DEFAULT_SEGMENTS: u32 = 8;
const MAX_SEGMENTS: u32 = 32;
const MIN_SEGMENT_BYTES: u64 = 1024 * 1024;
const RETRY_DELAYS_SECS: [u64; 5] = [1, 2, 4, 8, 16];

struct SegmentWork {
    index: u32,
    start: u64,
    end: u64,
    downloaded: u64,
    active: bool,
    done: bool,
}

fn initial_segments(size: u64, count: u32) -> Vec<SegmentWork>;
fn claim_next(segments: &mut [SegmentWork]) -> Option<usize>;
fn split_largest(segments: &mut Vec<SegmentWork>) -> Option<usize>;
fn segments_cover(segments: &[SegmentWork], size: u64) -> bool;
```

Rules:

- Segment count is `job.options().segments.unwrap_or(DEFAULT_SEGMENTS).clamp(1, MAX_SEGMENTS)`, reduced to at most `size` segments.
- `initial_segments` splits `[0, size)` into equal, contiguous, inclusive ranges.
- `claim_next` returns the first segment that is neither done nor active and marks it active.
- `split_largest` picks the segment with the largest remaining bytes. It refuses when remaining is below `2 * MIN_SEGMENT_BYTES`. The split point is the midpoint of the remaining range but never below the current write position. The target's `end` becomes `split_at - 1`; a new segment `[split_at, old_end]` is appended and returned. This works for active segments: their worker notices the shortened `end` before writing past it and discards extra response bytes.
- `run` uses the segmented path when `job.supports_ranges()` and `job.total_bytes()` is `Some(size)` with `size > 0`; otherwise it uses the M1 single-stream path.
- Load persisted segments with `ctx.store.list_segments(job.id())`. Reuse them only when `segments_cover` accepts them; otherwise clear and start fresh.
- Create/open `<name>.part` with write access and `set_len(size)` to preallocate. Do not truncate a resumed file.
- Spawn `count` workers in a `JoinSet`, each with its own file handle. Each worker loops: claim a segment, request `Range: bytes={position}-{end}`, stream until its current end or the range ends, then try `split_largest`; claim the new half, otherwise claim the next pending segment; exit when nothing is claimable.
- Send `If-Range` with the saved ETag, or Last-Modified when there is no ETag, whenever the job has saved progress. A `200 OK` answer to such a request sets a shared restart flag: stop every worker, clear segments, truncate the part file, and restart once from zero. A second `200` fails with `BackendError::RangeIgnored`.
- Retry request errors, 408/429/5xx statuses and body-read errors up to five times after the first attempt, sleeping 1, 2, 4, 8 and 16 seconds. After a body-read error, resume from the last written offset. Other 4xx statuses fail immediately.
- Before each write, check `is_cancelled()` then `is_paused()`. On either, flush, persist segments, and return `Paused`/`Cancelled`. Leave `<name>.part` in place.
- Apply `ctx.limiter.acquire(chunk_len)` before writing each chunk in both paths.
- Progress reports the sum of all segment `downloaded` values; the total is the known size.
- Persist a segment snapshot at most once per second while running and once on every exit. Use `Store::replace_segments`; a store error fails the run.
- On completion: flush and drop file handles, rename `.part` to the target path, `clear_segments`, and return `Completed`.
- The single-stream path reports `(0, total)` after removing `.part` when paused, so a resumed single-stream job starts from zero.

Add unit tests inside `http.rs` for: initial ranges cover the file; splitting halves the largest remaining range without overlap; ranges below 2 MiB are not split. Unit tests must not use `unwrap()` or `expect()`.

#![allow(clippy::unwrap_used, clippy::expect_used)]

use std::net::SocketAddr;
use std::sync::Arc;
use std::sync::atomic::{AtomicUsize, Ordering};
use std::time::Duration;

use axum::Router;
use axum::body::Body;
use axum::extract::State;
use axum::http::{StatusCode, header};
use axum::response::{IntoResponse, Response};
use axum::routing::get;
use odm_core::{Engine, JobOptions, JobState, Store};
use tempfile::TempDir;
use tokio::net::TcpListener;

const PAYLOAD: &[u8] = b"odm queue payload";

struct TestServer {
    addr: SocketAddr,
    active: Arc<AtomicUsize>,
    max_active: Arc<AtomicUsize>,
    handle: tokio::task::JoinHandle<()>,
}

#[derive(Clone)]
struct ServerState {
    active: Arc<AtomicUsize>,
    max_active: Arc<AtomicUsize>,
}

impl TestServer {
    async fn start() -> Self {
        let active = Arc::new(AtomicUsize::new(0));
        let max_active = Arc::new(AtomicUsize::new(0));
        let state = ServerState {
            active: active.clone(),
            max_active: max_active.clone(),
        };
        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
        let addr = listener.local_addr().unwrap();
        let app = Router::new()
            .route("/file.bin", get(file))
            .with_state(state);
        let handle = tokio::spawn(async move {
            axum::serve(listener, app).await.unwrap();
        });
        Self {
            addr,
            active,
            max_active,
            handle,
        }
    }

    fn url(&self) -> String {
        format!("http://{}/file.bin", self.addr)
    }
}

impl Drop for TestServer {
    fn drop(&mut self) {
        self.handle.abort();
    }
}

async fn file(State(state): State<ServerState>) -> Response {
    let active = state.active.fetch_add(1, Ordering::SeqCst) + 1;
    state.max_active.fetch_max(active, Ordering::SeqCst);
    tokio::time::sleep(Duration::from_millis(30)).await;
    state.active.fetch_sub(1, Ordering::SeqCst);

    (
        StatusCode::OK,
        [
            (header::CONTENT_LENGTH, PAYLOAD.len().to_string()),
            (header::CONTENT_TYPE, "application/octet-stream".to_string()),
        ],
        Body::from(PAYLOAD.to_vec()),
    )
        .into_response()
}

#[tokio::test]
async fn run_queue_respects_the_concurrency_limit() {
    let server = TestServer::start().await;
    let store = Store::open_in_memory().unwrap();
    let engine = Engine::new(store).unwrap();
    let dir = TempDir::new().unwrap();

    let mut ids = Vec::new();
    for _ in 0..4 {
        ids.push(
            engine
                .add(
                    &server.url(),
                    dir.path().to_path_buf(),
                    JobOptions::default(),
                )
                .unwrap()
                .id(),
        );
    }
    engine.store().set_queue_limit(1, 2).unwrap();

    let results = engine.run_queue("default").await.unwrap();
    assert_eq!(results.len(), 4);
    assert!(results.iter().all(|result| result.is_ok()));

    for id in ids {
        let job = engine.store().get_job(id).unwrap();
        assert_eq!(job.state(), JobState::Completed);
    }
    assert!(server.max_active.load(Ordering::SeqCst) <= 2);
}
